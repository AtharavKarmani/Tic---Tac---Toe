import java.util.Random;
import java.util.Scanner;

public class TicTacToe {

    static char[][] board = {
        {' ', ' ', ' '},
        {' ', ' ', ' '},
        {' ', ' ', ' '}
    };

    static char userSymbol, computerSymbol;
    static boolean isUserTurn;

    public static void main(String[] args) {
        performToss();
        printBoard();

        // Example: one move (you can later extend to loop)
        int slot = getUserSlot();
        int[] pos = getRowCol(slot);

        if (isValidMove(pos[0], pos[1])) {
            placeMove(pos[0], pos[1], userSymbol);
        } else {
            System.out.println("Invalid move!");
        }

        printBoard();
    }

    // 🔹 UC2: Toss
    public static void performToss() {
        Random random = new Random();
        Scanner sc = new Scanner(System.in);

        if (random.nextInt(2) == 0) {
            System.out.println("You won the toss!");
            System.out.print("Choose X or O: ");
            userSymbol = sc.next().toUpperCase().charAt(0);
            computerSymbol = (userSymbol == 'X') ? 'O' : 'X';
            isUserTurn = true;
        } else {
            System.out.println("Computer won the toss!");
            computerSymbol = (random.nextInt(2) == 0) ? 'X' : 'O';
            userSymbol = (computerSymbol == 'X') ? 'O' : 'X';
            isUserTurn = false;
        }

        System.out.println("User: " + userSymbol + ", Computer: " + computerSymbol);
    }

    // 🔹 UC3: Input
    public static int getUserSlot() {
        Scanner sc = new Scanner(System.in);
        int slot;

        while (true) {
            System.out.print("Enter slot (1-9): ");
            if (sc.hasNextInt()) {
                slot = sc.nextInt();
                if (slot >= 1 && slot <= 9) {
                    return slot;
                }
            } else {
                sc.next(); // clear invalid
            }
            System.out.println("Invalid input!");
        }
    }

    // 🔹 UC4: Convert
    public static int[] getRowCol(int slot) {
        int row = (slot - 1) / 3;
        int col = (slot - 1) % 3;
        return new int[]{row, col};
    }

    // 🔹 UC5: Validate
    public static boolean isValidMove(int row, int col) {
        if (row < 0 || row > 2 || col < 0 || col > 2) {
            return false;
        }
        return board[row][col] == ' ';
    }

    // 🔹 UC6: Place Move
    public static void placeMove(int row, int col, char symbol) {
        board[row][col] = symbol;
    }

    // 🔹 Extra: Print Board
    public static void printBoard() {
        System.out.println("\nBoard:");
        for (int i = 0; i < 3; i++) {
            System.out.println(" " + board[i][0] + " | " + board[i][1] + " | " + board[i][2]);
            if (i < 2) System.out.println("---|---|---");
        }
        System.out.println();
    }
}
